using namespace std;


class Advisory {
public:
    virtual void giveAdvice() = 0; 
    virtual ~Advisory() {}
};


class Crop : public Advisory {
protected:
    string name;
    double area;
    string season;
    double yieldPerAcre;

    
    static int cropCount;

public:
    
    Crop() : name("Unknown"), area(0.0), season("Unknown"), yieldPerAcre(0.0) {
        cropCount++;
    }

    
    Crop(string n, double a, string s, double y) : name(n), area(a), season(s), yieldPerAcre(y) {
        
        if (a < 0 || y < 0) {
            throw invalid_argument("Area and yield per acre cannot be negative!");
        }
        cropCount++;
    }

    
    virtual ~Crop() {
        cropCount--;
    }

    
    virtual double calculateTotalYield() const {
        return area * yieldPerAcre;
    }

    
    virtual void display() const {
        cout << left << setw(15) << name 
             << setw(10) << area 
             << setw(12) << season 
             << setw(15) << yieldPerAcre 
             << setw(15) << calculateTotalYield();
    }

    
    bool operator>(const Crop& other) const {
        return this->calculateTotalYield() > other.calculateTotalYield();
    }

    
    string getName() const { return name; }
    double getArea() const { return area; }
    string getSeason() const { return season; }
    double getYieldPerAcre() const { return yieldPerAcre; }


    static int getCropCount() {
        return cropCount;
    }
};


int Crop::cropCount = 0;


// 2. Derived class CerealCrop using inheritance
class CerealCrop : public Crop {
private:
    string fertilizerType; // Specific attribute

public:
    CerealCrop(string n, double a, string s, double y, string f) 
        : Crop(n, a, s, y), fertilizerType(f) {}

    void display() const override {
        Crop::display();
        cout << setw(20) << ("Fertilizer: " + fertilizerType) << endl;
    }

    // 6. Implementation of pure virtual function
    void giveAdvice() override {
        cout << "[Advice - " << name << "]: Apply " << fertilizerType 
             << " during the initial growth stage. Ensure uniform watering.\n";
    }
};



class VegetableCrop : public Crop {
private:
    int irrigationFrequency; // Specific attribute (in days)

public:
    VegetableCrop(string n, double a, string s, double y, int freq) 
        : Crop(n, a, s, y), irrigationFrequency(freq) {}

    void display() const override {
        Crop::display();
        cout << setw(20) << ("Irrigate every " + to_string(irrigationFrequency) + " days") << endl;
    }

    
    void giveAdvice() override {
        cout << "[Advice - " << name << "]: Maintain strict irrigation cycles every " 
             << irrigationFrequency << " days to prevent root diseases.\n";
    }
};


// 7. File Handling: Save crop data to file
void saveToFile(const vector<Crop*>& farm, const string& filename) {
    ofstream outFile(filename);
    if (!outFile) {
        throw runtime_error("Error: Unable to open file for writing.");
    }
    for (const auto& crop : farm) {
        outFile << crop->getName() << "," << crop->getArea() << "," 
                << crop->getSeason() << "," << crop->getYieldPerAcre() << "\n";
    }
    outFile.close();
}


void readFromFile(const string& filename) {
    ifstream inFile(filename);
    if (!inFile) {
        cout << "No file records found to read.\n";
        return;
    }
    cout << "\n--- Reading Saved Crop Data from File ---\n";
    string line;
    while (getline(inFile, line)) {
        cout << line << endl;
    }
    inFile.close();
}
