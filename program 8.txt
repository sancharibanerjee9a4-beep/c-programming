#include <iostream>
#include <string>
#include <iomanip>
using namespace std;

// ---------------- Class Student ----------------
class Student {
protected:
    int rollNo;
    char name[40];

public:
    void getStudentData() {
        cout << "Enter roll no : ";
        cin >> rollNo;
        cout << "Enter name : ";
        cin.ignore();
        cin.getline(name, 40);
    }

    void showStudentData() {
        cout << "Roll No : " << rollNo << "\n";
        cout << "Name : " << name << "\n";
    }
};

// ---------------- Derived class Exam ----------------
// Exam publicly inherits Student
class Exam : public Student {
protected:
    float marks[6]; // marks in 6 subjects

public:
    void getExamData() {
        cout << "\nEnter marks in 6 subjects (out of 100 each):\n";
        for (int i = 0; i < 6; i++) {
            cout << "  Subject " << (i + 1) << " : ";
            cin >> marks[i];
        }
    }

    void showExamData() {
        cout << "\nMarks obtained:\n";
        for (int i = 0; i < 6; i++) {
            cout << "  Subject " << (i + 1) << " : " << marks[i] << "\n";
        }
    }
};

// ---------------- Derived class Result (Level 3) ----------------
// Result publicly inherits Exam, which inherits Student -> MULTILEVEL INHERITANCE
class Result : public Exam {
private:
    float totalMarks;
    float percentage;
    char grade;

public:
    void calculateResult() {
        totalMarks = 0;
        for (int i = 0; i < 6; i++) {
            totalMarks += marks[i];
        }
        percentage = totalMarks / 6.0f; // 6 subjects, out of 100 each

        if (percentage >= 80) grade = 'A';
        else if (percentage >= 70) grade = 'B';
        else if (percentage >= 60) grade = 'C';
        else if (percentage >= 50) grade = 'D';
        else grade = 'F';
    }

    void displayResult() {
        cout << "\n------------------ RESULT SHEET ------------------\n";
        cout << "--------------------------------------------------\n";
        showStudentData(); // from Student
        showExamData();    // from Exam
        cout << fixed << setprecision(2);
        cout << "Total Marks : " << totalMarks << " / 600\n";
        cout << "Percentage  : " << percentage << " %\n";
        cout << "Grade       : " << grade << "\n";
        cout << "--------------------------------------------------\n";
    }
};

int main() {
    int n;
    cout << "Enter number of students: ";
    cin >> n;

    Result list[100];

    for (int i = 0; i < n; i++) {
        cout << "\nEnter details for student " << (i + 1) << ":\n";
        list[i].getStudentData();
        list[i].getExamData();
        list[i].calculateResult();
    }

    for (int i = 0; i < n; i++) {
        list[i].displayResult();
    }

    return 0;
}
